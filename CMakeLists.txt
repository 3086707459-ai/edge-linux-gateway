cmake_minimum_required(VERSION 3.16)
project(edge_linux_gateway VERSION 1.0.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This project uses Linux epoll, timerfd, signalfd and termios")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EDGE_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(EDGE_WERROR "Treat warnings as errors" ON)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
if(TARGET SQLite3::SQLite3)
  set(EDGE_SQLITE_TARGET SQLite3::SQLite3)
else()
  set(EDGE_SQLITE_TARGET SQLite::SQLite3)
endif()
add_library(edge_core src/config.cpp src/modbus.cpp src/serial.cpp src/storage.cpp src/http.cpp)
target_include_directories(edge_core PUBLIC include)
target_link_libraries(edge_core PUBLIC ${EDGE_SQLITE_TARGET} Threads::Threads)
target_compile_definitions(edge_core PUBLIC _GNU_SOURCE)
target_compile_options(edge_core PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(EDGE_WERROR)
  target_compile_options(edge_core PUBLIC -Werror)
endif()
if(EDGE_SANITIZE)
  target_compile_options(edge_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
  target_link_options(edge_core PUBLIC -fsanitize=address,undefined -no-pie)
endif()
add_executable(edge-gateway src/main.cpp src/gateway.cpp)
target_link_libraries(edge-gateway PRIVATE edge_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(edge-tests tests/test_core.cpp)
  target_link_libraries(edge-tests PRIVATE edge_core)
  add_test(NAME core COMMAND edge-tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:edge-gateway>)
  set_tests_properties(integration PROPERTIES TIMEOUT 60)
endif()
include(GNUInstallDirs)
install(TARGETS edge-gateway RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES config/gateway.conf DESTINATION ${CMAKE_INSTALL_SYSCONFDIR}/edge-gateway)
